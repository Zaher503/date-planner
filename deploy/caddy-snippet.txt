# NAS Caddyfile block for dates.zkocr.com (Phase 6). Modeled on food.zkocr.com.
# Live on the NAS since 2026-10-11 (backup Caddyfile.bak-20261011-dates). Back up before editing
# (heredoc rewrite, no sed -i), then reload Caddy.
#
# - POST /api/add: the iPhone Shortcut, checked by the app with X-Dates-Key (no Authentik).
# - POST /api/add-image: a screenshot from the Shortcut, same key, up to 10 MB.
# - /cal/*: calendar feed, checked by the app with a secret token in the path (no Authentik).
# - Everything else: Authentik forward_auth (group "dates"), which sets X-Authentik-Username.
# The passthrough routes strip any identity header a client sends, because the app trusts
# that header from the NAS.

dates.zkocr.com {
    @add {
        method POST
        path /api/add
    }
    handle @add {
        request_body {
            max_size 16KB
        }
        reverse_proxy 192.168.0.103:8820 192.168.0.172:8820 {
            header_up -X-Authentik-Username
            header_up -X-Authentik-Name
            lb_policy first
            lb_try_duration 3s
            health_uri /healthz
            health_interval 10s
            health_timeout 2s
        }
    }
    @addimage {
        method POST
        path /api/add-image
    }
    handle @addimage {
        request_body {
            max_size 11MB
        }
        reverse_proxy 192.168.0.103:8820 192.168.0.172:8820 {
            header_up -X-Authentik-Username
            header_up -X-Authentik-Name
            lb_policy first
            lb_try_duration 3s
            health_uri /healthz
            health_interval 10s
            health_timeout 2s
        }
    }
    handle /cal/* {
        reverse_proxy 192.168.0.103:8820 192.168.0.172:8820 {
            header_up -X-Authentik-Username
            header_up -X-Authentik-Name
            lb_policy first
            lb_try_duration 3s
            health_uri /healthz
            health_interval 10s
            health_timeout 2s
        }
    }
    handle {
        import authentik_auth
        reverse_proxy 192.168.0.103:8820 192.168.0.172:8820 {
            lb_policy first
            lb_try_duration 3s
            health_uri /healthz
            health_interval 10s
            health_timeout 2s
        }
    }
}
